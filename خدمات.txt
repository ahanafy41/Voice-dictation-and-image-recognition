function getGreeting()
    local hour = tonumber(os.date("%H"))
    if hour < 12 then
        return "Good morning"
    elseif hour < 18 then
        return "Good afternoon"
    else
        return "Good evening"
    end
end

function speakDateTime(context)
    local currentDate = os.date("%A, %B %d, %Y")
    local currentTime = os.time()
    local currentTimeInMillis = currentTime * 1000 -- Convert seconds to milliseconds
    local formattedTime = SimpleDateFormat("hh:mm:ss.SSS a").format(Date(currentTimeInMillis))

    local batteryIntent = context.registerReceiver(nil, IntentFilter(Intent.ACTION_BATTERY_CHANGED))
    local level = batteryIntent.getIntExtra(BatteryManager.EXTRA_LEVEL, -1)
    local scale = batteryIntent.getIntExtra(BatteryManager.EXTRA_SCALE, -1)
    local batteryPercent = math.floor((level / scale) * 100)

    -- Get greeting based on the time of day
    local greeting = getGreeting()

    -- Speak the message
    local message = greeting .. "! Today is " .. currentDate .. ". The time is " .. formattedTime .. ". The battery level is " .. batteryPercent .. " percent."
    context.speak(message)
end
